function replaceBetween(
  text,
  start,
  end,
  replacement,
  label
) {
  const a = text.indexOf(start);

  if (a < 0) {
    die(`${label}: start marker not found`);
  }

  const b = text.indexOf(
    end,
    a + start.length
  );

  if (b < 0) {
    die(`${label}: end marker not found`);
  }

  return (
    text.slice(0, a) +
    replacement +
    "\n" +
    text.slice(b)
  );
}
